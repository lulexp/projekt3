vector<string> Parser::tokenize(string line) {
	vector<string> tokens;
	string current;
	bool isReadingArgument = 0;
	bool isReadingOption = 0;

	for (char c : line) {

			if (c == '"' && isReadingArgument == 0)

Command* Parser::parseNoArguments(vector<string> tokens) {
	if (tokens[0] == "time") {
		return new Time();
	}

	if (tokens[0] == "date") {
		return new Date();
	}
}
